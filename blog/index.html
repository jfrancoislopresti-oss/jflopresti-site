(function(){ const TILT_MAX = 10; const cards = document.querySelectorAll('.abo-card, .blog-card, .block'); cards.forEach(card => { card.addEventListener('mousemove', e => { const rect = card.getBoundingClientRect(); const cx = rect.left + rect.width / 2; const cy = rect.top + rect.height / 2; const dx = (e.clientX - cx) / (rect.width / 2); const dy = (e.clientY - cy) / (rect.height / 2); const rx = -dy * TILT_MAX; const ry = dx * TILT_MAX; card.style.transform = `perspective(800px) rotateX(${rx}deg) rotateY(${ry}deg) translateZ(10px)`; card.style.boxShadow = `${-ry * 1.2}px ${rx * 1.2}px 28px rgba(232,84,26,0.22), 0 6px 32px rgba(27,58,92,0.12)`; }); card.addEventListener('mouseleave', () => { card.style.transform = 'perspective(800px) rotateX(0deg) rotateY(0deg) translateZ(0)'; card.style.boxShadow = ''; card.style.transition = 'transform 0.4s ease, box-shadow 0.4s ease'; }); card.addEventListener('mouseenter', () => { card.style.transition = 'transform 0.1s ease, box-shadow 0.1s ease'; }); }); // Parallaxe titres const lbls = document.querySelectorAll('.lbl, h1, h2'); window.addEventListener('scroll', () => { const sy = window.scrollY; lbls.forEach(el => { const rect = el.getBoundingClientRect(); const offset = (rect.top + rect.height / 2 - window.innerHeight / 2) * -0.04; el.style.transform = `translateY(${offset}px)`; }); }, {passive:true}); })();
